  # STAGE 3: Smoke Tests (after deploy)
  # ============================================
  - stage: SmokeTests
    displayName: 'Post-Deployment Smoke Tests'
    dependsOn: Deploy
    condition: succeeded()
    jobs:
      - job: SmokeTest
        displayName: 'Run Smoke Tests'
        pool:
          vmImage: 'ubuntu-latest'
        steps:
          - script: |
              set -euo pipefail
              echo "Verifying webhook endpoint..."
              FUNC_URL="https://$(functionAppName).azurewebsites.net/api/chatwoot-webhook"
              PAYLOAD='{"event":"conversation_updated"}'

              # Build signature header if webhook secret is configured
              SIG_HEADER=""
              if [ -n "${CHATWOOT_WEBHOOK_SECRET:-}" ]; then
                SIGNATURE=$(echo -n "$PAYLOAD" | openssl dgst -sha256 -hmac "$CHATWOOT_WEBHOOK_SECRET" | awk '{print $2}')
                SIG_HEADER="-H"
                SIG_EXTRA="X-Chatwoot-Signature:$SIGNATURE"
              fi

              HTTP_STATUS=$(curl -s -o /tmp/response.json -w "%{http_code}" \
                -X POST "$FUNC_URL" \
                -H "Content-Type: application/json" \
                ${SIG_HEADER:+"$SIG_HEADER" "$SIG_EXTRA"} \
                -d "$PAYLOAD" --max-time 30)

              echo "Webhook smoke test: HTTP $HTTP_STATUS"
              cat /tmp/response.json 2>/dev/null || true
              if [ "$HTTP_STATUS" = "200" ]; then
                echo "Smoke test passed"
              else
                echo "Smoke test failed (expected 200, got $HTTP_STATUS)"
                exit 1
              fi
            displayName: 'Verify webhook endpoint'
            env:
              CHATWOOT_WEBHOOK_SECRET: $(chatwootWebhookSecret)

          - script: |
              set -euo pipefail
              FUNC_URL="https://$(functionAppName).azurewebsites.net/api/chatwoot-webhook"
              if [ -n "${CW_FUNCTION_KEY:-}" ]; then
                FUNC_URL="${FUNC_URL}?code=${CW_FUNCTION_KEY}"
              fi

              check() {
                local label="$1"
                local payload="$2"
                local expect_skipped="$3"
                local resp
                resp=$(curl -sf -X POST "$FUNC_URL" \
                  -H "Content-Type: application/json" \
                  -d "$payload" --max-time 30 2>/dev/null || echo '{}')
                echo "[$label] $resp"
                if [ "$expect_skipped" = "true" ]; then
                  echo "$resp" | python3 -c "import sys,json; d=json.load(sys.stdin); assert d.get('skipped') or d.get('ok'), 'expected skipped/ok'" \
                    && echo "[$label] PASS" || { echo "[$label] FAIL"; exit 1; }
                fi
              }

              # Non-message event -> skipped (no agent call)
              check "event-gate" \
                '{"event":"conversation_updated","conversation":{"id":99901,"status":"pending"},"inbox":{"id":4}}' \
                "true"

              # Outgoing message (agent/bot) -> skipped (only incoming processed)
              check "outgoing-gate" \
                '{"event":"message_created","message_type":1,"content":"bot reply","private":false,"conversation":{"id":99902,"status":"open"},"inbox":{"id":4},"sender":{"id":1,"type":"user"}}' \
                "true"

              # Private note -> skipped
              check "private-gate" \
                '{"event":"message_created","message_type":0,"content":"internal note","private":true,"conversation":{"id":99903,"status":"open"},"inbox":{"id":4},"sender":{"id":1,"type":"contact"}}' \
                "true"

              echo "Spec-compliance smoke tests passed."
            displayName: 'Spec-compliance smoke tests'
            env:
              CW_FUNCTION_KEY: $(CW_FUNCTION_KEY)

  # ============================================
  # STAGE 4: Notify (Telegram, after deploy + smoke)
  # ============================================
  # Format dictated by Yasha (2026-05-03): see /home/shovalbe/docs/tlg_notification 1.sh.
  # Single rich Markdown message on success or failure. SUCCESS includes commit message,
  # author, branch, build ID + URL. FAILURE flips icon and label, otherwise identical.
  # Pipeline secrets: tgBotToken (e.g. "bot1234:abc..."), tgChatId.
  - stage: Notify
    displayName: 'Notify (Telegram)'
    dependsOn:
      - Deploy
      - SmokeTests
    # Run only on master deploys (where Deploy actually executes). PR-validation
    # builds skip Deploy + SmokeTests, so notifying for them would be noise.
    # `always()` within this scope ensures we still notify on Deploy/Smoke FAIL,
    # but the outer eq() check skips non-master branches outright.
    condition: |
      and(
        eq(variables['Build.SourceBranch'], 'refs/heads/master'),
        in(dependencies.Deploy.result, 'Succeeded', 'Failed'),
        in(dependencies.SmokeTests.result, 'Succeeded', 'Failed', 'Skipped')
      )
    jobs:
      - job: TelegramNotify
        displayName: 'Send Telegram build notification'
        pool:
          vmImage: 'ubuntu-latest'
        steps:
          - checkout: self
          - script: |
              set -euo pipefail

              # Aggregate stage outcomes — STATUS is FAILED if any of Deploy/SmokeTests failed.
              # Note: deployResult / smokeResult are ENV vars set via the `env:` block below
              # (runtime expressions). Bash reads them as $deployResult / $smokeResult — NOT
              # the ADO macro syntax $(...), which would be interpreted by bash as command
              # substitution and fail with "command not found" (build 11923 hit this bug).
              if [ "${deployResult:-}" = "Succeeded" ] && [ "${smokeResult:-Skipped}" != "Failed" ]; then
                STATUS="SUCCESS"
              else
                STATUS="FAILED"
              fi

              if [ "$STATUS" = "FAILED" ]; then
                ICON="[FAIL]"
                LABEL="Build FAILED"
              else
                ICON="[OK]"
                LABEL="Build Succeeded"
              fi

              COMMIT_MESSAGE=$(git log --format=%B -n 1 2>/dev/null | head -5)
              DEPLOY_TIME=$(date -u '+%Y-%m-%d %H:%M UTC')
              BUILD_URL="https://dev.azure.com/Corp-domain/Corp-AI/_build/results?buildId=$(Build.BuildId)"

              MSG="${ICON} *${LABEL}*
              *Pipeline:* \`cs-agents-ci\`
              *Branch:* \`$(Build.SourceBranchName)\`
              *Author:* $(Build.RequestedFor)
              *Commit:* ${COMMIT_MESSAGE}
              *Time:* ${DEPLOY_TIME}
              *Build ID:* $(Build.BuildId)
              [View Pipeline Run](${BUILD_URL})"

              RESPONSE=$(curl -s -w "\n%{http_code}" -X POST \
                "https://api.telegram.org/${TG_BOT_TOKEN}/sendMessage" \
                -d "chat_id=${TG_CHAT_ID}" \
                -d "parse_mode=Markdown" \
                --data-urlencode "text=${MSG}")

              HTTP_CODE=$(echo "$RESPONSE" | tail -n1)
              BODY=$(echo "$RESPONSE" | head -n -1)

              if [ "$HTTP_CODE" != "200" ]; then
                echo "Telegram notification failed (HTTP $HTTP_CODE): $BODY"
                # Don't fail the pipeline if only the notification failed.
                exit 0
              fi
              echo "Telegram notification sent (HTTP $HTTP_CODE)"
            displayName: 'Send Telegram notification'
            env:
              TG_BOT_TOKEN: $(tgBotToken)
              TG_CHAT_ID: $(tgChatId)
              deployResult: $[ stageDependencies.Deploy.result ]
              smokeResult: $[ stageDependencies.SmokeTests.result ]

  # ============================================
  # STAGE 6: Eval regression gate (P1-4) - ADVISORY
  # ============================================
  # Runs run_live_eval.py --gate against the blessed baseline (evals/data/baseline_passk.json).
  # Auth is via the AIServices API key (the stage lists it with the SP's Contributor role and
  # exports AZURE_AI_FOUNDRY_KEY), NOT the SP's AAD token, because the SP lacks the Foundry-User
  # data role (that AAD path returned 403). So the gate now RUNS.
  # It stays ADVISORY (continueOnError + exit 0) for ONE remaining reason: it evaluates the
  # DEPLOYED agent (not this PR's Function code) with trials=1 nano run-variance, so it must not
  # hard-block a code merge. To make it BLOCKING: re-bless the baseline at --trials 3 and drop the
  # advisory exit. Bless/re-bless: run_live_eval.py --trials 3 --update-baseline evals/data/baseline_passk.json
  - stage: YashaEvalGate
    displayName: 'Eval Regression Gate (seekapa live, advisory)'
    dependsOn: []
    condition: eq(variables['Build.Reason'], 'PullRequest')
    jobs:
      - job: evalgate
        displayName: 'run_live_eval --gate vs blessed baseline (advisory)'
        continueOnError: true
        pool:
          vmImage: 'ubuntu-latest'
        timeoutInMinutes: 20
        steps:
          - checkout: self
            fetchDepth: 1
          - task: UsePythonVersion@0
            displayName: 'Use Python $(pythonVersion)'
            inputs:
              versionSpec: '$(pythonVersion)'
              addToPath: true
          - script: |
              python -m pip install --upgrade pip uv
              uv pip install --system 'azure-ai-projects>=2.0.0' 'azure-identity>=1.15.0' 'openai>=1.40.0'
            displayName: 'Install Foundry eval SDKs'
          - task: AzureCLI@2
            displayName: 'Eval gate (regression vs baseline)'
            inputs:
              azureSubscription: '$(azureServiceConnection)'
              scriptType: bash
              scriptLocation: inlineScript
              addSpnToEnvironment: true
              inlineScript: |
                set -uo pipefail
                # Key auth: the SP has Contributor (it deploys the app) so it can list the
                # AIServices key; run_live_eval then calls the agent with the key, avoiding the
                # Foundry-User data role the SP lacks (that is why AAD returned 403).
                export AZURE_AI_FOUNDRY_KEY="$(az cognitiveservices account keys list -n brn-azai -g AZAI_group --query key1 -o tsv 2>/dev/null || true)"
                if [ -z "${AZURE_AI_FOUNDRY_KEY:-}" ]; then
                  echo "[eval-gate] Could not fetch the brn-azai key (SP lacks listKeys). Advisory."
                  exit 0
                fi
                if [ ! -f evals/data/baseline_passk.json ]; then
                  echo "[eval-gate] No blessed baseline yet; advisory. Bless via --update-baseline."
                  exit 0
                fi
                python evals/scripts/run_live_eval.py --gate --baseline evals/data/baseline_passk.json --trials 1
                rc=$?
                if [ $rc -ne 0 ]; then
                  echo "[eval-gate] ADVISORY (not blocking this code PR): run_live_eval exited $rc."
                  echo "[eval-gate] rc=1 = regression vs baseline (or trials=1 nano variance). Re-bless at --trials 3 to make this blocking."
                fi
                exit 0
